var got string

type stub struct{ a2asrv.RequestHandler }

func (stub) SubscribeToTask(_ context.Context, r *a2a.SubscribeToTaskRequest) iter.Seq2[a2a.Event, error] {
	got = "SubscribeToTask(" + string(r.ID) + ")"
	return func(y func(a2a.Event, error) bool) { y(nil, a2a.ErrTaskNotFound) }
}
func (stub) SendMessage(context.Context, *a2a.SendMessageRequest) (a2a.SendMessageResult, error) {
	got = "SendMessage"
	return nil, a2a.ErrInvalidParams
}
func (stub) GetTask(_ context.Context, r *a2a.GetTaskRequest) (*a2a.Task, error) {
	got = "GetTask(" + string(r.ID) + ")"
	return nil, a2a.ErrTaskNotFound
}

var restPaths = []string{"/message:send", "/message:stream", "/tasks", "/tasks/", "/extendedAgentCard"}

func mux8080() http.Handler {
	card := &a2a.AgentCard{Name: "x"}
	m := http.NewServeMux()
	m.Handle(a2asrv.WellKnownAgentCardPath, a2asrv.NewStaticAgentCardHandler(card))
	rest := a2asrv.NewRESTHandler(stub{})
	for _, p := range restPaths {
		m.Handle(p, rest)
	}
	m.Handle("/", a2asrv.NewJSONRPCHandler(stub{}))
	return root(m)
}

func root(a http.Handler) http.Handler {
	r := http.NewServeMux()
	r.HandleFunc("/healthz", func(w http.ResponseWriter, _ *http.Request) { got = "healthz" })
	r.HandleFunc("/control/inject", func(w http.ResponseWriter, _ *http.Request) { got = "control" })
	r.HandleFunc("/control/reset", func(w http.ResponseWriter, _ *http.Request) { got = "control" })
	r.Handle("/", a)
	return r
}

type nopCloser struct{ *bytes.Reader }

func (nopCloser) Close() error { return nil }
func ioNop(b []byte) nopCloser  { return nopCloser{bytes.NewReader(b)} }
